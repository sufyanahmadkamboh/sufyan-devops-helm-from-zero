  labels:
    {{- include "demo-app.labels" . | nindent 4 }}

  {{- with .Values.ingress.className }}
  ingressClassName: {{ . }}
  {{- end }}

    {{- range .Values.ingress.tls }}

        {{- range .hosts }}
        - {{ . | quote }}
        {{- end }}

    {{- end }}
  rules:
    {{- range .Values.ingress.hosts }}
    - host: {{ .host | quote }}
      http:
        paths:
          {{- range .paths }}
          - path: {{ .path }}
            pathType: {{ .pathType }}
            backend:
              service:
                name: {{ include "demo-app.fullname" $ }}
                port:
                  number: {{ $.Values.service.port }}
          {{- end }}
    {{- end }}
{{- end }}
